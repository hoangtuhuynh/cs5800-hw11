package vendingmachine;

public class DispensingSnackState implements StateOfVendingMachine {
    private final VendingMachine vendingMachine;

    public DispensingSnackState(VendingMachine vendingMachine) {
        this.vendingMachine = vendingMachine;
    }

    @Override
    public void selectSnack(String snackName) {
        System.out.println("Currently dispensing. Please wait.");
    }

    @Override
    public void insertMoney(double amount) {
        System.out.println("Currently dispensing. Please wait.");
    }

    @Override
    public void dispenseSnack() {
        Snack selected = vendingMachine.getSelectedSnack();
        if (selected == null) {
            System.out.println("No snack selected.");
            vendingMachine.resetTransaction();
            vendingMachine.setState(vendingMachine.getIdleState());
            return;
        }

        boolean dispensed = vendingMachine.getSnackDispenser().handle(vendingMachine, selected.getName());
        if (dispensed) {
            double change = vendingMachine.getInsertedMoney() - selected.getPrice();
            if (change > 0) {
                System.out.println("Returning change: $" + String.format("%.2f", change));
            }
        } else {
            System.out.println("Could not dispense " + selected.getName() + ". Returning $"
                    + String.format("%.2f", vendingMachine.getInsertedMoney()));
        }

        vendingMachine.resetTransaction();
        vendingMachine.setState(vendingMachine.getIdleState());
        System.out.println("State -> Idle");
    }

    @Override
    public String toString() {
        return "Dispensing Snack";
    }
}

package vendingmachine;

public class IdleState implements StateOfVendingMachine {
    private final VendingMachine vendingMachine;

    public IdleState(VendingMachine vendingMachine) {
        this.vendingMachine = vendingMachine;
    }

    @Override
    public void selectSnack(String snackName) {
        Snack snack = vendingMachine.getSnack(snackName);
        if (snack == null) {
            System.out.println("Snack not found: " + snackName);
            return;
        }
        vendingMachine.setSelectedSnack(snack);
        System.out.println("Selected " + snack.getName() + " ($" + String.format("%.2f", snack.getPrice()) + ")");
        vendingMachine.setState(vendingMachine.getWaitingForMoneyState());
        System.out.println("State -> Waiting for Money");
    }

    @Override
    public void insertMoney(double amount) {
        System.out.println("Select a snack before inserting money.");
    }

    @Override
    public void dispenseSnack() {
        System.out.println("Cannot dispense while idle. Select a snack first.");
    }

    @Override
    public String toString() {
        return "Idle";
    }
}

========== src/main/java/vendingmachine/Snack.java ==========
package vendingmachine;

public class Snack {
    private final String name;
    private final double price;
    private int quantity;

    public Snack(String name, double price, int quantity) {
        if (name == null || name.isBlank()) {
            throw new IllegalArgumentException("name cannot be null or blank");
        }
        if (price < 0) {
            throw new IllegalArgumentException("price cannot be negative");
        }
        if (quantity < 0) {
            throw new IllegalArgumentException("quantity cannot be negative");
        }
        this.name = name;
        this.price = price;
        this.quantity = quantity;
    }

    public String getName() {
        return name;
    }

    public double getPrice() {
        return price;
    }

    public int getQuantity() {
        return quantity;
    }

    public boolean isAvailable() {
        return quantity > 0;
    }

    public void decrementQuantity() {
        if (quantity <= 0) {
            throw new IllegalStateException(name + " is out of stock");
        }
        quantity--;
    }

    @Override
    public String toString() {
        return name + " ($" + String.format("%.2f", price) + ", qty=" + quantity + ")";
    }
}

========== src/main/java/vendingmachine/SnackDispenseHandler.java ==========
package vendingmachine;

public abstract class SnackDispenseHandler {
    private SnackDispenseHandler nextHandler;

    public SnackDispenseHandler setNext(SnackDispenseHandler nextHandler) {
        this.nextHandler = nextHandler;
        return nextHandler;
    }

    public SnackDispenseHandler getNextHandler() {
        return nextHandler;
    }

    public final boolean handle(VendingMachine machine, String snackName) {
        if (canHandle(snackName)) {
            System.out.println(getClass().getSimpleName() + " handling request for " + snackName);
            doDispense(machine, snackName);
            return true;
        }

        System.out.println(getClass().getSimpleName() + " passing " + snackName + " to next handler");
        if (nextHandler != null) {
            return nextHandler.handle(machine, snackName);
        }

        System.out.println("No handler found for snack: " + snackName);
        return false;
    }

    protected abstract boolean canHandle(String snackName);

    protected void doDispense(VendingMachine machine, String snackName) {
        Snack snack = machine.getSnack(snackName);
        snack.decrementQuantity();
        System.out.println("Dispensed " + snack.getName() + ". Remaining quantity: " + snack.getQuantity());
    }
}

package vendingmachine;

import java.util.Collection;
import java.util.LinkedHashMap;
import java.util.Map;

public class VendingMachine {
    private final Map<String, Snack> snacks = new LinkedHashMap<>();
    private final StateOfVendingMachine idleState;
    private final StateOfVendingMachine waitingForMoneyState;
    private final StateOfVendingMachine dispensingSnackState;
    private final SnackDispenseHandler snackDispenser;

    private StateOfVendingMachine currentState;
    private Snack selectedSnack;
    private double insertedMoney;

    public VendingMachine(SnackDispenseHandler snackDispenser) {
        if (snackDispenser == null) {
            throw new IllegalArgumentException("snackDispenser cannot be null");
        }
        this.snackDispenser = snackDispenser;
        this.idleState = new IdleState(this);
        this.waitingForMoneyState = new WaitingForMoneyState(this);
        this.dispensingSnackState = new DispensingSnackState(this);
        this.currentState = idleState;
    }

    public void addSnack(Snack snack) {
        if (snack == null) {
            throw new IllegalArgumentException("snack cannot be null");
        }
        snacks.put(snack.getName().toLowerCase(), snack);
    }

    public Snack getSnack(String name) {
        if (name == null) {
            return null;
        }
        return snacks.get(name.toLowerCase());
    }

    public Collection<Snack> getSnacks() {
        return snacks.values();
    }

    public void selectSnack(String snackName) {
        currentState.selectSnack(snackName);
    }

    public void insertMoney(double amount) {
        currentState.insertMoney(amount);
    }

    public void dispenseSnack() {
        currentState.dispenseSnack();
    }

    public StateOfVendingMachine getCurrentState() {
        return currentState;
    }

    public void setState(StateOfVendingMachine state) {
        if (state == null) {
            throw new IllegalArgumentException("state cannot be null");
        }
        this.currentState = state;
    }

    public StateOfVendingMachine getIdleState() {
        return idleState;
    }

    public StateOfVendingMachine getWaitingForMoneyState() {
        return waitingForMoneyState;
    }

    public StateOfVendingMachine getDispensingSnackState() {
        return dispensingSnackState;
    }

    public SnackDispenseHandler getSnackDispenser() {
        return snackDispenser;
    }

    public Snack getSelectedSnack() {
        return selectedSnack;
    }

    public void setSelectedSnack(Snack selectedSnack) {
        this.selectedSnack = selectedSnack;
    }

    public double getInsertedMoney() {
        return insertedMoney;
    }

    public void addInsertedMoney(double amount) {
        this.insertedMoney += amount;
    }

    public void resetInsertedMoney() {
        this.insertedMoney = 0;
    }

    public void resetTransaction() {
        this.selectedSnack = null;
        this.insertedMoney = 0;
    }

    public void printInventory() {
        System.out.println("--- Inventory ---");
        for (Snack snack : snacks.values()) {
            System.out.println(snack);
        }
    }
}

package vendingmachine;

public class VendingMachineDriver {
    public static void main(String[] args) {
        SnackDispenseHandler chain = buildSnackChain();
        VendingMachine machine = new VendingMachine(chain);

        machine.addSnack(new Snack("Coke", 1.50, 3));
        machine.addSnack(new Snack("Pepsi", 1.50, 3));
        machine.addSnack(new Snack("Cheetos", 1.25, 2));
        machine.addSnack(new Snack("Doritos", 1.25, 2));
        machine.addSnack(new Snack("KitKat", 1.00, 2));
        machine.addSnack(new Snack("Snickers", 1.00, 1));

        System.out.println("=== Vending Machine ready (State: " + machine.getCurrentState() + ") ===");
        machine.printInventory();

        System.out.println("\n=== Purchase Coke with exact change ===");
        machine.selectSnack("Coke");
        machine.insertMoney(1.50);

        System.out.println("\n=== Purchase Pepsi with extra money (get change) ===");
        machine.selectSnack("Pepsi");
        machine.insertMoney(2.00);

        System.out.println("\n=== Purchase Cheetos with partial then remaining money ===");
        machine.selectSnack("Cheetos");
        machine.insertMoney(0.50);
        machine.insertMoney(0.75);

        System.out.println("\n=== Purchase Doritos ===");
        machine.selectSnack("Doritos");
        machine.insertMoney(1.25);

        System.out.println("\n=== Purchase KitKat ===");
        machine.selectSnack("KitKat");
        machine.insertMoney(1.00);

        System.out.println("\n=== Purchase last Snickers (quantity becomes 0) ===");
        machine.selectSnack("Snickers");
        machine.insertMoney(1.00);

        System.out.println("\n=== Attempt another Snickers while quantity is 0 ===");
        machine.selectSnack("Snickers");
        machine.insertMoney(1.00);

        System.out.println("\n=== Attempt insert money while idle ===");
        machine.insertMoney(1.00);

        System.out.println();
        machine.printInventory();
        System.out.println("Final state: " + machine.getCurrentState());
    }

    static SnackDispenseHandler buildSnackChain() {
        SnackDispenseHandler coke = new CokeDispenseHandler();
        SnackDispenseHandler pepsi = new PepsiDispenseHandler();
        SnackDispenseHandler cheetos = new CheetosDispenseHandler();
        SnackDispenseHandler doritos = new DoritosDispenseHandler();
        SnackDispenseHandler kitKat = new KitKatDispenseHandler();
        SnackDispenseHandler snickers = new SnickersDispenseHandler();

        coke.setNext(pepsi);
        pepsi.setNext(cheetos);
        cheetos.setNext(doritos);
        doritos.setNext(kitKat);
        kitKat.setNext(snickers);

        return coke;
    }
}

package vendingmachine;

public class WaitingForMoneyState implements StateOfVendingMachine {
    private final VendingMachine vendingMachine;

    public WaitingForMoneyState(VendingMachine vendingMachine) {
        this.vendingMachine = vendingMachine;
    }

    @Override
    public void selectSnack(String snackName) {
        Snack snack = vendingMachine.getSnack(snackName);
        if (snack == null) {
            System.out.println("Snack not found: " + snackName);
            return;
        }
        vendingMachine.setSelectedSnack(snack);
        vendingMachine.resetInsertedMoney();
        System.out.println("Changed selection to " + snack.getName()
                + " ($" + String.format("%.2f", snack.getPrice()) + "). Inserted money reset.");
    }

    @Override
    public void insertMoney(double amount) {
        if (amount <= 0) {
            System.out.println("Insert a positive amount of money.");
            return;
        }

        vendingMachine.addInsertedMoney(amount);
        Snack selected = vendingMachine.getSelectedSnack();
        System.out.println("Inserted $" + String.format("%.2f", amount)
                + ". Total inserted: $" + String.format("%.2f", vendingMachine.getInsertedMoney()));

        if (vendingMachine.getInsertedMoney() < selected.getPrice()) {
            double remaining = selected.getPrice() - vendingMachine.getInsertedMoney();
            System.out.println("Need $" + String.format("%.2f", remaining) + " more for " + selected.getName());
            return;
        }

        if (!selected.isAvailable()) {
            System.out.println(selected.getName() + " is out of stock. Returning $"
                    + String.format("%.2f", vendingMachine.getInsertedMoney()));
            vendingMachine.resetTransaction();
            vendingMachine.setState(vendingMachine.getIdleState());
            System.out.println("State -> Idle");
            return;
        }

        vendingMachine.setState(vendingMachine.getDispensingSnackState());
        System.out.println("State -> Dispensing Snack");
        vendingMachine.dispenseSnack();
    }

    @Override
    public void dispenseSnack() {
        System.out.println("Insert enough money before dispensing.");
    }

    @Override
    public String toString() {
        return "Waiting for Money";
    }
}

        assertTrue(chain.getNextHandler() instanceof PepsiDispenseHandler);
        assertTrue(chain.getNextHandler().getNextHandler() instanceof CheetosDispenseHandler);

        assertTrue(chain.getNextHandler().getNextHandler().getNextHandler().getNextHandler()

    }

    @Test
    void cokeHandlerDispensesCoke() {
        assertTrue(chain.handle(machine, "Coke"));
        assertEquals(1, machine.getSnack("Coke").getQuantity());
    }

    @Test
    void snickersRequestWalksFullChain() {
        assertTrue(chain.handle(machine, "Snickers"));
